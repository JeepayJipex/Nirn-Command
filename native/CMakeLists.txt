cmake_minimum_required(VERSION 3.25)
project(NirnCommandNative VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(directxtk CONFIG REQUIRED)
find_package(CommonLibSSE CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_commonlibsse_plugin(NirnCommandNative NAME "NirnCommandNative" AUTHOR "Nirn Command" VERSION ${PROJECT_VERSION} USE_ADDRESS_LIBRARY SOURCES main.cpp discovery.cpp storage.cpp)
target_compile_features(NirnCommandNative PRIVATE cxx_std_23)
target_precompile_headers(NirnCommandNative PRIVATE PCH.h)
target_link_libraries(NirnCommandNative PRIVATE CommonLibSSE::CommonLibSSE nlohmann_json::nlohmann_json)

if(MSVC)
  target_compile_options(NirnCommandNative PRIVATE /utf-8)
endif()
enable_testing()
add_executable(discovery_test discovery.cpp discovery_test.cpp)
target_compile_features(discovery_test PRIVATE cxx_std_23)
target_link_libraries(discovery_test PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(discovery_test PRIVATE /utf-8)
endif()
add_test(NAME discovery COMMAND discovery_test)

add_executable(storage_test storage.cpp storage_test.cpp)
target_compile_features(storage_test PRIVATE cxx_std_23)
target_link_libraries(storage_test PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(storage_test PRIVATE /utf-8)
endif()
add_test(NAME storage COMMAND storage_test)
