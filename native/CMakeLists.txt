cmake_minimum_required(VERSION 3.25)
project(NirnCommandNative VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(directxtk CONFIG REQUIRED)
find_package(CommonLibSSE CONFIG REQUIRED)
add_commonlibsse_plugin(NirnCommandNative NAME "NirnCommandNative" AUTHOR "Nirn Command" VERSION ${PROJECT_VERSION} USE_ADDRESS_LIBRARY SOURCES main.cpp)
target_compile_features(NirnCommandNative PRIVATE cxx_std_23)
target_precompile_headers(NirnCommandNative PRIVATE PCH.h)
target_link_libraries(NirnCommandNative PRIVATE CommonLibSSE::CommonLibSSE)
